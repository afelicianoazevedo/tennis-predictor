-- ============================================================
-- TENNIS PREDICTOR
-- Database V1
-- Supabase / PostgreSQL

-- ============================================================
-- 1. PLAYERS
-- ============================================================

create table public.players (
    id bigint generated by default as identity primary key,

    api_id bigint not null unique,

    name text not null,
    country text,
    gender text,
    ranking integer,
    ranking_points numeric,
    hand text,
    birth_date date,
    photo_url text,

    created_at timestamptz not null default now(),
    updated_at timestamptz not null default now()
);


-- ============================================================
-- 2. TOURNAMENTS
-- ============================================================

create table public.tournaments (
    id bigint generated by default as identity primary key,

    api_id bigint not null unique,

    name text not null,
    category text,
    surface text,
    country text,

    start_date date,
    end_date date,

    created_at timestamptz not null default now(),
    updated_at timestamptz not null default now()
);


-- ============================================================
-- 3. MATCHES
-- ============================================================

create table public.matches (
    id bigint generated by default as identity primary key,

    api_id bigint not null unique,

    tournament_id bigint references public.tournaments(id)
        on delete set null,

    scheduled_at timestamptz,

    round text,

    status text not null default 'scheduled',

    player1_id bigint references public.players(id)
        on delete set null,

    player2_id bigint references public.players(id)
        on delete set null,

    winner_id bigint references public.players(id)
        on delete set null,

    score text,

    best_of integer,

    surface text,

    created_at timestamptz not null default now(),
    updated_at timestamptz not null default now(),

    constraint matches_different_players
        check (
            player1_id is null
            or player2_id is null
            or player1_id <> player2_id
        )
);


-- ============================================================
-- 4. MATCH PREDICTIONS
-- ============================================================

create table public.match_predictions (
    id bigint generated by default as identity primary key,

    match_id bigint not null references public.matches(id)
        on delete cascade,

    created_at timestamptz not null default now(),

    player1_probability numeric(5,2),
    player2_probability numeric(5,2),

    confidence_score numeric(5,2),

    confidence_level text,

    predicted_winner_id bigint references public.players(id)
        on delete set null,

    model_version text,

    result text,

    was_correct boolean
);


-- ============================================================
-- 5. ODDS
-- ============================================================

create table public.odds (
    id bigint generated by default as identity primary key,

    match_id bigint not null references public.matches(id)
        on delete cascade,

    captured_at timestamptz not null default now(),

    player1_odd numeric(10,3),
    player2_odd numeric(10,3),

    source text,
    market text
);


-- ============================================================
-- 6. API USAGE
-- ============================================================

create table public.api_usage (
    id bigint generated by default as identity primary key,

    usage_date date not null unique,

    requests_used integer not null default 0,

    requests_reserved integer not null default 0,

    last_request_at timestamptz,

    created_at timestamptz not null default now(),

    constraint api_usage_requests_used_positive
        check (requests_used >= 0),

    constraint api_usage_requests_reserved_positive
        check (requests_reserved >= 0)
);


-- ============================================================
-- 7. API REQUEST LOG
-- ============================================================

create table public.api_requests (
    id bigint generated by default as identity primary key,

    requested_at timestamptz not null default now(),

    endpoint text not null,

    parameters jsonb,

    http_status integer,

    success boolean not null default false,

    request_type text,

    response_time_ms integer,

    error_message text
);

-- ============================================================
-- UPDATED_AT TRIGGER
-- ============================================================

create or replace function public.set_updated_at()
returns trigger
language plpgsql
as $$
begin
    new.updated_at = now();
    return new;
end;
$$;


create trigger players_updated_at
before update on public.players
for each row
execute function public.set_updated_at();


create trigger tournaments_updated_at
before update on public.tournaments
for each row
execute function public.set_updated_at();


create trigger matches_updated_at
before update on public.matches
for each row
execute function public.set_updated_at();


-- ============================================================
-- API QUOTA FUNCTION
-- ============================================================

create or replace function public.get_api_usage(
    p_date date default current_date
)
returns integer
language sql
as $$
    select coalesce(
        (
            select requests_used
            from public.api_usage
            where usage_date = p_date
        ),
        0
    );
$$;


-- ============================================================
-- REGISTER API REQUEST
-- ============================================================

create or replace function public.register_api_request()
returns integer
language plpgsql
as $$
declare
    current_usage integer;
begin

    insert into public.api_usage (
        usage_date,
        requests_used,
        last_request_at
    )
    values (
        current_date,
        1,
        now()
    )
    on conflict (usage_date)
    do update
    set
        requests_used = public.api_usage.requests_used + 1,
        last_request_at = now();

    select requests_used
    into current_usage
    from public.api_usage
    where usage_date = current_date;

    return current_usage;

end;
$$;


-- ============================================================
-- CAN MAKE API REQUEST?
-- We use 90 as the normal operational limit.
-- 100 remains the absolute API limit.
-- ============================================================

create or replace function public.can_make_api_request()
returns boolean
language sql
as $$
    select public.get_api_usage(current_date) < 90;
$$;
